  <script>
    function handleAuth(e) {
      e.preventDefault();
      document.getElementById('auth-screen').classList.add('hidden');
      document.getElementById('dashboard').classList.remove('hidden');
    }

    function logout() {
      document.getElementById('dashboard').classList.add('hidden');
      document.getElementById('auth-screen').classList.remove('hidden');
    }

    function addTerminal(e) {
      e.preventDefault();
      const name = document.getElementById('pc-name').value;
      const ip = document.getElementById('pc-ip').value;
      
      const grid = document.getElementById('terminal-grid');
      const card = document.createElement('div');
      card.className = "bg-slate-950 border border-slate-800 rounded-lg p-4 relative";
      card.innerHTML = `
        <div class="flex items-center justify-between mb-2">
          <span class="font-bold text-sm text-white">${name}</span>
          <span class="text-[10px] font-mono text-slate-500">${ip}</span>
        </div>
        <div class="text-xs text-slate-400 mb-1">Active App / Site:</div>
        <div class="text-xs font-mono bg-slate-900 p-2 rounded text-slate-400 border border-slate-800 truncate mb-3">
          Desktop Idle
        </div>
        <div class="flex items-center justify-between text-[11px] text-slate-500">
          <span>Status: Standby</span>
          <span class="text-emerald-500">● Online</span>
        </div>
      `;
      grid.appendChild(card);
      
      document.getElementById('pc-count').innerText = grid.children.length;
      document.getElementById('pc-name').value = '';
      document.getElementById('pc-ip').value = '';
    }

    function addRestriction(e) {
      e.preventDefault();
      const input = document.getElementById('restricted-url');
      const val = input.value.trim();
      if (!val) return;

      const container = document.getElementById('restricted-tags');
      const tag = document.createElement('span');
      tag.className = "bg-red-500/10 text-red-400 border border-red-500/20 text-xs px-2.5 py-1 rounded-full flex items-center gap-1";
      tag.innerHTML = `${val} <button onclick="this.parentElement.remove()" class="hover:text-white">×</button>`;
      container.appendChild(tag);
      input.value = '';
    }
  </script>
